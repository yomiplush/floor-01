cmake_minimum_required(VERSION 3.20)
project(Floor01 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(QRENCODE REQUIRED IMPORTED_TARGET libqrencode)
add_executable(floor01 main.cpp)
target_link_libraries(floor01 PRIVATE Qt6::Widgets Qt6::Network PkgConfig::SDL2 PkgConfig::QRENCODE)
qt_add_resources(floor01 web PREFIX "/" FILES web/index.html)
qt_add_resources(floor01 samples PREFIX "/" FILES
    samples/kick.wav samples/snare.wav samples/hat.wav samples/open.wav
    samples/perc.wav samples/bass808.wav samples/cymbal.wav)
target_compile_options(floor01 PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS floor01 RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(address_test tests/address_test.cpp)
    target_include_directories(address_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(address_test PRIVATE Qt6::Network)
    add_test(NAME address_test COMMAND address_test)
endif()
